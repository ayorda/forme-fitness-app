```bash
ffmpeg -i original.MP4 -ss 0 -t 10.5 -map 0:v:0 \
  -vf 'scale=854:480,fps=24' -c:v libx264 -preset slow -crf 30 \
  -profile:v main -level 3.1 -pix_fmt yuv420p -an -map_metadata -1 \
  -movflags +faststart parallel-bar-dips-v1.mp4

  -row-mt 1 -an -map_metadata -1 parallel-bar-dips-v1.webm
```
